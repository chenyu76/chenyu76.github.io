\newcommand{\circledtext}[3]{%
  \tikz[baseline=(X.base)]{%
    \node[circle, draw=none, fill=#2, inner sep=2pt] (X) {%
      \textcolor{#1}{#3}%
    };%
  }%
}

\tcbset{
  my flex/.style={
    enhanced,
    % nobeforeafter,       % 取消前后间距
    % hbox,                % 宽度由内部内容决定
    arc=8pt,
    boxrule=1.2pt,
    colback=#1!0,
    colframe=#1,
    fonttitle=\bfseries,
    boxsep=0pt, top=4pt, bottom=4pt, left=4pt, right=4pt
  },
  % 子盒子专用样式
  sub flex/.style={
    enhanced,
    % nobeforeafter,
    % hbox,
    arc=4pt,             % 子盒子圆角小一点
    boxrule=1pt,
    colback=#1!0,
    colframe=#1, % !80
    fonttitle=\small\bfseries,
    center title,
    % halign=center,       % 内部文字水平居中
    valign=center,
    % 稍微收紧子盒子的内边距
    boxsep=0pt, top=4pt, bottom=4pt, left=4pt, right=4pt
  },
  my flex/.default=gray,
  sub flex/.default=gray
}
